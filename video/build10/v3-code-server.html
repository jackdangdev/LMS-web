server.js — the process entry point

Four jobs, in order: env → database → listen → graceful shutdown. No listen() inside app.js, so tests can import it.

require('dotenv').config();
const app = require('./src/app');
const { connectDB, disconnectDB } = require('./src/config/db');
const PORT = process.env.PORT || 5000;
const start = async () => {
await connectDB(); // 1. database first
const server = app.listen(PORT, () => { // 2. then listen
console.log(`[api] EduNova LMS running on :${PORT}/api`);
});
const shutdown = async (signal) => { // 3. graceful stop
server.close(async () => {
await disconnectDB(); // no half-written data
process.exit(0);
});
};
['SIGINT', 'SIGTERM'].forEach((sig) => process.on(sig, () => shutdown(sig)));
};
start();
Why it matters
Tests import app without a port; deploys stop cleanly; config never hardcodes credentials.
