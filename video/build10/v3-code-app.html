app.js — the middleware pipeline

Layer order is the security model: security → parsing → logging → routes → 404 → error funnel.

const app = express();
app.use(helmet()); // security headers
app.use(cors({ origin: CLIENT_ORIGINS, credentials: true }));
app.use(express.json({ limit: '1mb' })); // body parsing
if (process.env.NODE_ENV !== 'test') app.use(morgan('dev'));
app.get('/api/health', (_req, res) => res.json({ status: 'UP' }));
app.use('/api/auth', authRoutes); // four routers,
app.use('/api/users', userRoutes); // one JSON contract
app.use('/api/courses', courseRoutes);
app.use('/api/enrollments', enrollmentRoutes);
app.use(notFound); // unmatched request?
app.use(errorHandler); // one error funnel:
// 400/401/403/404/409 -> JSON
Uniform envelope
Success: { success, data }
Failure: { success:false, message } with the right status code — the client handles errors one way.
